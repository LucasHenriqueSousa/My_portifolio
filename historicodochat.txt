/**
 * ==========================================================
 * menu.js
 * Controle da Navbar
 * ==========================================================

let menuButton;
let navigation;
let navLinks;

/**
 * Abre o menu
 */
function openMenu() {

    menuButton.classList.add("active");
    navigation.classList.add("open");

    menuButton.setAttribute("aria-expanded", "true");

    document.body.classList.add("menu-open");

}

/**
 * Fecha o menu
 */
function closeMenu() {

    menuButton.classList.remove("active");
    navigation.classList.remove("open");

    menuButton.setAttribute("aria-expanded", "false");

    document.body.classList.remove("menu-open");

}

/**
 * Alterna o menu
 */
function toggleMenu() {

    if (navigation.classList.contains("open")) {

        closeMenu();

    } else {

        openMenu();

    }

}

/**
 * Fecha clicando em um link
 */
function closeOnLinkClick() {

    navLinks.forEach((link) => {

        link.addEventListener("click", () => {

            closeMenu();

        });

    });

}

/**
 * Fecha clicando fora
 */
function closeOnOutsideClick(event) {

    if (
        navigation.contains(event.target) ||
        menuButton.contains(event.target)
    ) {
        return;
    }

    closeMenu();

}

/**
 * Fecha com ESC
 */
function closeOnEscape(event) {

    if (event.key !== "Escape") {

        return;

    }

    closeMenu();

}

/**
 * Navegação por teclado
 */
function keyboardAccessibility(event) {

    if (
        event.key === "Enter" ||
        event.key === " "
    ) {

        event.preventDefault();

        toggleMenu();

    }

}

/**
 * Inicialização
 */
export function initMenu() {

    menuButton = document.querySelector(".menu-toggle");

    navigation = document.querySelector(".nav-links");

    if (!menuButton || !navigation) {

        console.error("[MENU] Elementos não encontrados.");

        return;

    }

    navLinks = navigation.querySelectorAll("a");

    menuButton.setAttribute("aria-expanded", "false");

    menuButton.setAttribute("aria-label", "Abrir menu");

    menuButton.addEventListener("click", toggleMenu);

    menuButton.addEventListener("keydown", keyboardAccessibility);

    document.addEventListener("click", closeOnOutsideClick);

    document.addEventListener("keydown", closeOnEscape);

    closeOnLinkClick();

}

Estrutura: este módulo é responsável por revelar elementos quando entram na área visível da tela, utilizando apenas um IntersectionObserver.

Performance: os elementos são observados apenas uma vez (unobserve()), evitando processamento desnecessário.

Dependências: utiliza as classes definidas em animations.css (.hidden, .hidden-left, .hidden-right, .hidden-scale).

assets/js/reveal.js
/**
 * ==========================================================
 * reveal.js
 * Reveal Animation utilizando IntersectionObserver
 * ==========================================================
 */

const REVEAL_SELECTORS = [
    ".hidden",
    ".hidden-left",
    ".hidden-right",
    ".hidden-scale"
];

/**
 * Aplica delay automático baseado na posição do elemento
 */
function applyAutoDelay(elements) {

    elements.forEach((element, index) => {

        const delay = (index % 6) * 120;

        element.style.transitionDelay = `${delay}ms`;

    });

}

/**
 * Revela elemento conforme sua classe
 */
function revealElement(element) {

    if (element.classList.contains("hidden")) {
        element.classList.add("show");
    }

    if (element.classList.contains("hidden-left")) {
        element.classList.add("show-left");
    }

    if (element.classList.contains("hidden-right")) {
        element.classList.add("show-right");
    }

    if (element.classList.contains("hidden-scale")) {
        element.classList.add("show-scale");
    }

}

/**
 * Cria IntersectionObserver
 */
function createObserver() {

    return new IntersectionObserver(

        (entries, observer) => {

            entries.forEach((entry) => {

                if (!entry.isIntersecting) {

                    return;

                }

                revealElement(entry.target);

                observer.unobserve(entry.target);

            });

        },

        {
            root: null,
            rootMargin: "0px 0px -80px 0px",
            threshold: 0.15
        }

    );

}

/**
 * Observa todos os elementos
 */
function observeElements(observer) {

    REVEAL_SELECTORS.forEach((selector) => {

        const elements = document.querySelectorAll(selector);

        if (!elements.length) {

            return;

        }

        applyAutoDelay([...elements]);

        elements.forEach((element) => {

            observer.observe(element);

        });

    });

}

/**
 * Inicialização
 */
export function initReveal() {

    if (!("IntersectionObserver" in window)) {

        console.warn(
            "[Reveal] IntersectionObserver não suportado. Revelando todos os elementos."
        );

        REVEAL_SELECTORS.forEach((selector) => {

            document.querySelectorAll(selector).forEach((element) => {

                revealElement(element);

            });

        });

        return;

    }

    const observer = createObserver();

    observeElements(observer);

}
